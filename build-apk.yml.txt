name: Construire l'APK Android

on:
  workflow_dispatch: {}
  push:
    branches: [main, master]

jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true

      - name: Générer le projet Android
        run: flutter create . --org com.tonalis --project-name tonalis --platforms=android

      - name: Autorisations Android (Internet pour les polices, voix système)
        run: |
          python3 - <<'PY'
          p = "android/app/src/main/AndroidManifest.xml"
          s = open(p, encoding="utf-8").read()
          add = (
              '<uses-permission android:name="android.permission.INTERNET"/>\n'
              '    <queries>\n'
              '        <intent>\n'
              '            <action android:name="android.intent.action.TTS_SERVICE"/>\n'
              '        </intent>\n'
              '    </queries>\n'
              '    <application'
          )
          assert "<application" in s
          open(p, "w", encoding="utf-8").write(s.replace("<application", add, 1))
          PY

      - run: flutter pub get

      - name: Analyse (rapport seulement)
        run: flutter analyze
        continue-on-error: true

      - run: flutter build apk --release

      - uses: actions/upload-artifact@v4
        with:
          name: tonalis-apk
          path: build/app/outputs/flutter-apk/app-release.apk
